CUSTOM GUARDRAIL:

  from src.incident_iq.services.guardrails_service import BaseGuard, GuardType, GuardResult, GuardailCheckResult

  class MyGuard(BaseGuard):
      def check(self, data):
          if not self._validate(data):
              return GuardailCheckResult(
                  guard_type=GuardType.POLICY_ENFORCEMENT,
                  guard_name=self.name,
                  result=GuardResult.FAIL,
                  message="Failed",
                  severity="high"
              )
          return GuardailCheckResult(
              guard_type=GuardType.POLICY_ENFORCEMENT,
              guard_name=self.name,
              result=GuardResult.PASS,
              message="Passed",
              severity="low"
          )

  guardrails.register_guardrail(MyGuard(), scope='custom')
